Complete recovery function

rsa_lab/crypto.py · recover_private_key() · lines 13–29

def recover_private_key(n, e, factor):
    if type(factor) is not int or not 1 < factor < n or n % factor:
        raise ValueError("Factor must be a nontrivial divisor of n")
    p = factor
    q = n // p
    if p == q or not isPrime(p) or not isPrime(q):
        raise ValueError("n must be a product of two distinct primes")

    # With both primes known, recover d by modular inversion.
    lambda_n = lcm(p - 1, q - 1)
    if gcd(e, lambda_n) != 1:
        raise ValueError("Public exponent is not coprime to lambda(n)")
    d = pow(e, -1, lambda_n)
    key = RSA.construct((n, e, d, p, q), consistency_check=True)
    if key.n != p * q or (e * d) % lambda_n != 1:
        raise ArithmeticError("Recovered key failed consistency checks")
    return key

def oaep_cipher(key, label=b""):
    # Use SHA-256 for both OAEP and MGF1.
    return PKCS1_OAEP.new(key, hashAlgo=SHA256,
                         mgfunc=lambda seed, size: PKCS1_OAEP.MGF1(seed, size, SHA256), label=label)


def decrypt_message(private, row):
    if (row.get("scheme"), row.get("hash"), row.get("mgf")) != ("RSAES-OAEP", "SHA-256", "MGF1-SHA-256"):
        raise ValueError("Only RSAES-OAEP / SHA-256 / MGF1-SHA-256 is supported")
    label = base64.b64decode(row["label_b64"], validate=True)
    ciphertext = base64.b64decode(row["ciphertext_b64"], validate=True)
    if len(ciphertext) != private.size_in_bytes():
        raise ValueError("Ciphertext length does not match the modulus length")
    return oaep_cipher(private, label).decrypt(ciphertext)

The recovered private key is passed to decrypt_message(). The code checks the declared scheme and ciphertext length, decodes the inputs, and uses PyCryptodome to decrypt with matching SHA-256, MGF1 and label parameters. OAEP decryption provides the plaintext; the separate evaluation compares it with the expected message.
