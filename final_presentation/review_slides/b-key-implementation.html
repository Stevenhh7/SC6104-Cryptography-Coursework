13def recover_private_key(n, e, factor):14    if type(factor) is not int or not 1 < factor < n or n % factor:15        raise ValueError("Expected a nontrivial divisor")16    p = factor17    q = n // p18    if p == q or not isPrime(p) or not isPrime(q):19        raise ValueError("Expected two distinct primes")22    lambda_n = lcm(p - 1, q - 1)23    if gcd(e, lambda_n) != 1:24        raise ValueError("e must be coprime to lambda(n)")25    d = pow(e, -1, lambda_n)26    key = RSA.construct((n, e, d, p, q), consistency_check=True)27    if key.n != p * q or (e * d) % lambda_n != 1:28        raise ArithmeticError("Key consistency check failed")29    return key

q = N / p   ·   λ(N) = lcm(p − 1, q − 1)   ·   d = e−1 mod λ(N)
